import java.awt.*;

class GridExample {
    public static void main(String[] args) {
        Frame f = new Frame("Calculator");

        f.setLayout(new GridLayout(4, 4));

        String b[] = {
            "1", "2", "3", "+",
            "4", "5", "6", "-",
            "7", "8", "9", "*",
            "0", "C", "=", "/"
        };

        for (int i = 0; i < 16; i++) {
            f.add(new Button(b[i]));
        }

        f.setSize(300, 300);
        f.setVisible(true);
    }
}
